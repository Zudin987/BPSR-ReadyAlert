#[derive(Clone,Copy)]struct DpsRowLayout{name_left:i32,name_right:i32,badge_left:i32,total_left:i32,total_right:i32,active_left:i32,active_right:i32,enc_left:i32,enc_right:i32,share_left:i32,share_right:i32}
fn dps_row_layout(r:RECT,show_imagines:bool,row:&DpsRow)->DpsRowLayout{let width=(r.right-r.left).max(1);let share_w=58;let total_w=84;let rate_w=if width>=760{96}else{84};let gap=4;let share_right=r.right-3;let share_left=share_right-share_w;let enc_right=share_left-gap;let enc_left=enc_right-rate_w;let active_right=enc_left-gap;let active_left=active_right-rate_w;let total_right=active_left-gap;let total_left=total_right-total_w;let name_left=r.left+25;let badges=if show_imagines&&!row.is_dead&&width>=700{row.imagines.len().min(2)as i32}else{0};let badge_span=badges*BADGE_W+badges.saturating_sub(1)*BADGE_GAP;let badge_left=(total_left-badge_span-7).max(name_left+80);let name_right=if badge_span>0{badge_left-6}else{total_left-8};DpsRowLayout{name_left,name_right,badge_left,total_left,total_right,active_left,active_right,enc_left,enc_right,share_left,share_right}}
fn dps_identity(row:&DpsRow)->String{let spec=if !row.subprofession_name.trim().is_empty(){row.subprofession_name.as_str()}else{profession_name(row.profession_id)};if spec.trim().is_empty(){row.name.clone()}else{format!("{} · {}",row.name,spec)}}
fn dps_spec(row:&DpsRow)->String{if !row.subprofession_name.trim().is_empty(){row.subprofession_name.clone()}else{profession_name(row.profession_id).to_string()}}
fn dps_score_pair(row:&DpsRow)->String{match(row.ability_score>0,row.illusion_break>0){(true,true)=>format!("{} +{}",score(row.ability_score),score(row.illusion_break)),(true,false)=>score(row.ability_score),(false,true)=>format!("+{}",score(row.illusion_break)),_=>String::new()}}
fn revive_status_text(row:&DpsRow,now:i64)->(String,u32){if row.revive_blocked_until_ms<0{return("REVIVE ?".into(),rgb(255,190,70));}if row.revive_blocked_until_ms>now{let left=row.revive_blocked_until_ms.saturating_sub(now)as u64;return(format!("REVIVE {}",format_countdown(left)),rgb(255,190,70));}("CAN REVIVE".into(),rgb(72,226,116))}
fn rate(total:i64,encounter_ms:u64)->f64{if total<=0{0.0}else{let seconds=(encounter_ms.max(1)as f64/1000.0).max(0.001);total as f64/seconds}}
fn encounter_rate(row:&DpsRow,mode:SortMode,encounter_ms:u64)->f64{match mode{SortMode::Damage=>row.dps,SortMode::Heal=>rate(row.healing,encounter_ms),SortMode::Tank=>rate(row.damage_taken,encounter_ms)}}
fn active_rate(row:&DpsRow,mode:SortMode)->f64{match mode{SortMode::Damage=>row.active_dps,SortMode::Heal=>row.active_hps,SortMode::Tank=>row.active_dtps}}
fn mode_metric(row:&DpsRow,mode:SortMode)->i64{match mode{SortMode::Damage=>row.damage,SortMode::Heal=>row.healing,SortMode::Tank=>row.damage_taken}.max(0)}
fn active_share(row:&DpsRow,mode:SortMode,settings:&FeatureSettings)->Option<(f64,u32)>{match mode{SortMode::Damage if settings.meter.show_damage_share=>Some((row.damage_share,rgb(244,102,102))),SortMode::Heal if settings.meter.show_healing_share=>Some((row.healing_share,rgb(87,220,133))),SortMode::Tank if settings.meter.show_tank_share=>Some((row.tank_share,rgb(92,169,245))),_=>None}}

fn snapshot_mode_total(snapshot:&DpsSnapshot,mode:SortMode)->i64{match mode{SortMode::Damage=>snapshot.total_damage,SortMode::Heal=>snapshot.total_healing,SortMode::Tank=>snapshot.total_damage_taken}}
fn mode_label(mode:SortMode)->&'static str{match mode{SortMode::Damage=>"DAMAGE",SortMode::Heal=>"HEALING",SortMode::Tank=>"TAKEN"}}
fn target_title(snapshot:&DpsSnapshot)->String{match snapshot.target.as_ref(){Some(t)if!t.name.trim().is_empty()=>t.name.clone(),Some(t)=>format!("Target #{}",t.entity_uuid),None=>"No active target".into()}}
fn target_hp(snapshot:&DpsSnapshot)->String{snapshot.target.as_ref().map(|t|if t.max_hp>0{format!("HP {:.1}%",(t.hp.max(0)as f64*100.0/t.max_hp as f64).clamp(0.0,100.0))}else{"HP ?".into()}).unwrap_or_else(||"HP —".into())}
fn capture_short(state:&State)->String{let short=state.capture_status.split('|').next().unwrap_or(state.capture_status.as_str()).trim();if short.is_empty(){"Capture".into()}else{short.to_string()}}

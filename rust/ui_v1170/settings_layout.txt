unsafe fn build_ui(hwnd: HWND, state: &mut SettingsState) {
    let nav = ["General", "Chat overlay", "Chat colors", "Speech", "Tabs & filters", "Sounds & logs", "Network", "Blocked users"];
    for (i, name) in nav.iter().enumerate() {
        create_button(hwnd, NAV_BASE + i as i32, name, 10, 14 + i as i32 * 38, 136, 30);
    }
    let version=create_static(hwnd,&format!("ReadyAlert  v{}",env!("CARGO_PKG_VERSION")),14,326,130,20);
    crate::ui_theme::set_font(version,crate::ui_theme::FontRole::Secondary);
    create_button(hwnd, ID_APPLY, "Apply", 610, 365, 80, 30);
    create_button(hwnd, ID_CLOSE, "Close", 700, 365, 80, 30);
    build_general(hwnd, state);
    build_overlay(hwnd, state);
    build_colors(hwnd, state);
    build_speech(hwnd, state);
    build_tabs(hwnd, state);
    build_sounds(hwnd, state);
    build_network(hwnd, state);
    build_blocked(hwnd, state);
}

fn detail_skills(state:&DetailState)->Vec<&SkillBreakdown>{
    let mut out:Vec<&SkillBreakdown>=match state.mode{
        DetailMode::Damage=>state.row.skills.iter().filter(|s|s.damage>0).collect(),
        DetailMode::Heal=>state.row.skills.iter().filter(|s|s.healing>0).collect(),
        _=>Vec::new(),
    };
    out.sort_by(|a,b|detail_skill_amount(b,state.mode).cmp(&detail_skill_amount(a,state.mode)));
    out
}
fn detail_skill_amount(skill:&SkillBreakdown,mode:DetailMode)->i64{match mode{DetailMode::Damage=>skill.damage,DetailMode::Heal=>skill.healing,_=>0}}
fn detail_mode_label(mode:DetailMode)->&'static str{match mode{DetailMode::Damage=>"Damage",DetailMode::Heal=>"Healing",DetailMode::Tank=>"Taken",DetailMode::Buffs=>"Buff Uptime",DetailMode::Deaths=>"Death Recap"}}
fn block_text(value:i64)->String{let pct=value as f64/100.0;let mut text=format!("{pct:.2}");while text.contains('.')&&text.ends_with('0'){text.pop();}if text.ends_with('.') {text.pop();}format!("{text}%")}
fn detail_biggest_hit(row:&DpsRow)->i64{row.taken_sources.iter().map(|x|x.max_value).max().unwrap_or(0)}
fn detail_biggest_absorb(row:&DpsRow)->i64{row.absorbed_sources.iter().map(|x|x.max_value).max().unwrap_or(0)}
#[derive(Clone,Copy)]struct TankView<'a>{row:&'a crate::model::TakenSourceBreakdown,shield:bool}
fn detail_taken_rows(state:&DetailState)->Vec<TankView<'_>>{
    let mut rows:Vec<TankView<'_>>=state.row.taken_sources.iter().map(|row|TankView{row,shield:false})
        .chain(state.row.absorbed_sources.iter().map(|row|TankView{row,shield:true})).collect();
    rows.sort_by(|a,b|b.row.damage.cmp(&a.row.damage).then_with(||b.row.max_value.cmp(&a.row.max_value)).then_with(||a.row.source_name.cmp(&b.row.source_name)));
    rows
}
#[derive(Clone)]struct DeathLine{header:bool,text:String,value:String,color:u32}
fn death_lines(state:&DetailState)->Vec<DeathLine>{
    let mut out=Vec::new();
    for recap in state.row.death_recaps.iter().rev(){
        let five_start=recap.at_ms.saturating_sub(5_000);
        let dmg5:i64=recap.events.iter().filter(|e|!e.is_heal&&e.at_ms>=five_start).map(|e|e.value).fold(0_i64,i64::saturating_add);
        let heal5:i64=recap.events.iter().filter(|e|e.is_heal&&e.at_ms>=five_start).map(|e|e.value).fold(0_i64,i64::saturating_add);
        out.push(DeathLine{header:true,text:format!("Death {} @ {}  |  last 5s DMG {}  HEAL {}",recap.death_no,format_time(recap.at_ms),compact(dmg5 as f64),compact(heal5 as f64)),value:format!("{} events",recap.events.len()),color:rgb(255,100,100)});
        let last_damage=recap.events.iter().rposition(|event|!event.is_heal);
        for(index,event)in recap.events.iter().enumerate().rev(){
            let before=recap.at_ms.saturating_sub(event.at_ms)as f64/1000.0;
            let kind=if event.is_heal{"HEAL"}else{"DMG"};
            let flags=match(event.crit,event.lucky){(true,true)=>" C+L",(true,false)=>" C",(false,true)=>" L",_=>""};
            let last=if Some(index)==last_damage{"  LAST HIT"}else{""};
            out.push(DeathLine{header:false,text:format!("-{before:.1}s  {kind}{flags}{last}  {}  |  {} [{}]",event.source_name,event.skill_name,event.skill_id),value:compact(event.value as f64),color:if event.is_heal{rgb(100,225,145)}else if Some(index)==last_damage{rgb(255,80,80)}else{rgb(255,120,120)}});
        }
    }
    out
}

struct ConsumablePopupState{parent:HWND}
unsafe fn create_consumable_popup(instance:HINSTANCE,parent:HWND)->HWND{
    let class=wide(CONSUMABLE_CLASS);let wc=WNDCLASSW{lpfnWndProc:Some(consumable_wnd_proc),hInstance:instance,hCursor:LoadCursorW(null_mut(),IDC_ARROW),lpszClassName:class.as_ptr(),..std::mem::zeroed()};
    if RegisterClassW(&wc)==0&&GetLastError()!=1410{return null_mut();}
    let state=Box::new(ConsumablePopupState{parent});let ptr=Box::into_raw(state);let title=wide("ReadyAlert Consumables");
    let hwnd=CreateWindowExW(WS_EX_TOOLWINDOW|WS_EX_LAYERED|WS_EX_TOPMOST|0x08000000,class.as_ptr(),title.as_ptr(),WS_POPUP,0,0,DPS_CONSUMABLE_POPUP_W,1,parent,null_mut(),instance,ptr.cast::<c_void>());
    if hwnd.is_null(){drop(Box::from_raw(ptr));return null_mut();}
    // Color-keyed transparent pixels mean there is no overlay background beside
    // the meter; only the circular F/S indicators are visible.
    SetLayeredWindowAttributes(hwnd,DPS_CONSUMABLE_KEY,0,1);ShowWindow(hwnd,SW_SHOW);hwnd
}
unsafe fn sync_consumable_popup(parent:HWND,state:&mut State){
    if state.kind!=Kind::Dps||state.consumable_hwnd.is_null()||IsWindow(state.consumable_hwnd)==0{return;}
    if state.collapsed{ShowWindow(state.consumable_hwnd,0);return;}
    let mut r:RECT=std::mem::zeroed();if GetWindowRect(parent,&mut r)==0{return;}let top=dps_rows_top();let h=(r.bottom-r.top-top).max(1);
    SetWindowPos(state.consumable_hwnd,HWND_TOPMOST,r.left-DPS_CONSUMABLE_POPUP_W-DPS_CONSUMABLE_POPUP_GAP,r.top+top,DPS_CONSUMABLE_POPUP_W,h,SWP_NOACTIVATE);ShowWindow(state.consumable_hwnd,SW_SHOW);
}
unsafe fn popup_status_at<'a>(state:&'a State,x:i32,y:i32,now:i64)->Option<(&'a ConsumableStatus,&'static str)>{
    if y<0{return None;}let screen_i=(y/DPS_ROW_H)as usize;let rows=meter_rows(state);let row=*rows.get(state.scroll+screen_i)?;
    let iy=screen_i as i32*DPS_ROW_H+((DPS_ROW_H-2-DPS_CONSUMABLE_ICON)/2).max(0);if y<iy||y>=iy+DPS_CONSUMABLE_ICON{return None;}
    let fx=2;let sx=fx+DPS_CONSUMABLE_ICON+DPS_CONSUMABLE_GAP;
    if x>=fx&&x<fx+DPS_CONSUMABLE_ICON{return active_consumable(row.food.as_ref(),now).map(|s|(s,"Food"));}
    if x>=sx&&x<sx+DPS_CONSUMABLE_ICON{return active_consumable(row.serum.as_ref(),now).map(|s|(s,"Serum"));}None
}
unsafe fn paint_consumable_popup(hwnd:HWND,state:&ConsumablePopupState){
    let parent_ptr=GetWindowLongPtrW(state.parent,GWLP_USERDATA)as *mut State;let mut ps:PAINTSTRUCT=std::mem::zeroed();let hdc=BeginPaint(hwnd,&mut ps);if hdc.is_null(){return;}
    let mut rc:RECT=std::mem::zeroed();GetClientRect(hwnd,&mut rc);fill(hdc,&rc,DPS_CONSUMABLE_KEY);
    if !parent_ptr.is_null(){let parent=&*parent_ptr;if !parent.collapsed{let now=now_ms();let rows=meter_rows(parent);let visible=(rc.bottom/DPS_ROW_H).max(0)as usize;
        for screen_i in 0..visible{let Some(row)=rows.get(parent.scroll+screen_i).copied()else{break;};let y=screen_i as i32*DPS_ROW_H+((DPS_ROW_H-2-DPS_CONSUMABLE_ICON)/2).max(0);let fx=2;let sx=fx+DPS_CONSUMABLE_ICON+DPS_CONSUMABLE_GAP;
            if let Some(food)=active_consumable(row.food.as_ref(),now){paint_consumable_ring(hdc,fx,y,"F",food,now);}if let Some(serum)=active_consumable(row.serum.as_ref(),now){paint_consumable_ring(hdc,sx,y,"S",serum,now);}
        }
    }}EndPaint(hwnd,&ps);
}
unsafe fn on_consumable_mouse_move(hwnd:HWND,state:&ConsumablePopupState,lparam:LPARAM){
    let x=lo_signed(lparam);let y=hi_signed(lparam);let parent_ptr=GetWindowLongPtrW(state.parent,GWLP_USERDATA)as *mut State;if parent_ptr.is_null(){return;}let parent=&mut*parent_ptr;let now=now_ms();
    let next=popup_status_at(parent,x,y,now).map(|(status,kind)|format!("{}: {} · {}",kind,status.name,consumable_remaining_text(status,now)));
    if next!=parent.hover_text{parent.hover_text=next;parent.hover_x=4;parent.hover_y=dps_rows_top()+y;InvalidateRect(state.parent,null(),0);}
    let mut track=TrackMouseEventData{cb_size:std::mem::size_of::<TrackMouseEventData>()as u32,flags:TME_LEAVE_,hwnd,hover_time:0};TrackMouseEvent(&mut track);
}
unsafe extern "system" fn consumable_wnd_proc(hwnd:HWND,msg:u32,_wparam:WPARAM,lparam:LPARAM)->LRESULT{
    if msg==WM_NCCREATE{let create=lparam as *const CREATESTRUCTW;if !create.is_null(){SetWindowLongPtrW(hwnd,GWLP_USERDATA,(*create).lpCreateParams as isize);}}
    let ptr=GetWindowLongPtrW(hwnd,GWLP_USERDATA)as *mut ConsumablePopupState;
    match msg{
        WM_ERASEBKGND=>1,
        WM_NCHITTEST=>{if ptr.is_null(){return -1;}let mut r:RECT=std::mem::zeroed();GetWindowRect(hwnd,&mut r);let sx=(lparam as i32 as i16)as i32;let sy=((lparam>>16)as i16)as i32;let x=sx-r.left;let y=sy-r.top;let parent_ptr=GetWindowLongPtrW((*ptr).parent,GWLP_USERDATA)as *mut State;if parent_ptr.is_null(){-1}else if popup_status_at(&*parent_ptr,x,y,now_ms()).is_some(){HTCLIENT as isize}else{-1}},
        WM_MOUSEMOVE=>{if !ptr.is_null(){on_consumable_mouse_move(hwnd,&*ptr,lparam);}0},
        WM_MOUSELEAVE_=>{if !ptr.is_null(){let parent_ptr=GetWindowLongPtrW((*ptr).parent,GWLP_USERDATA)as *mut State;if !parent_ptr.is_null(){(*parent_ptr).hover_text=None;InvalidateRect((*ptr).parent,null(),0);}}0},
        WM_PAINT=>{if !ptr.is_null(){paint_consumable_popup(hwnd,&*ptr);}0},
        WM_NCDESTROY=>{SetWindowLongPtrW(hwnd,GWLP_USERDATA,0);if !ptr.is_null(){drop(Box::from_raw(ptr));}DefWindowProcW(hwnd,msg,_wparam,lparam)},
        _=>DefWindowProcW(hwnd,msg,_wparam,lparam)
    }
}

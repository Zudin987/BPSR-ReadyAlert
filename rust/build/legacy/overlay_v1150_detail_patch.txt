unsafe fn detail_card(hdc:HDC,r:RECT,title:&str,lines:&[String]){fill(hdc,&r,rgb(22,29,35));fill(hdc,&RECT{left:r.left,top:r.top,right:r.left+3,bottom:r.bottom},rgb(66,211,190));SetTextColor(hdc,rgb(112,205,190));draw(hdc,title,RECT{left:r.left+10,top:r.top+3,right:r.right-7,bottom:r.top+20},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX);SetTextColor(hdc,rgb(228,235,239));for(i,text)in lines.iter().take(4).enumerate(){draw(hdc,text,RECT{left:r.left+10,top:r.top+20+i as i32*14,right:r.right-7,bottom:r.top+35+i as i32*14},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);}}
unsafe fn paint_detail(hwnd:HWND,state:&DetailState){
    let mut ps:PAINTSTRUCT=std::mem::zeroed();let hdc=BeginPaint(hwnd,&mut ps);if hdc.is_null(){return;}
    let mut rc:RECT=std::mem::zeroed();GetClientRect(hwnd,&mut rc);fill(hdc,&rc,rgb(15,19,24));let font=if state.font.is_null(){GetStockObject(DEFAULT_GUI_FONT)}else{state.font};SelectObject(hdc,font);SetBkMode(hdc,TRANSPARENT as i32);
    let row=&state.row;paint_popup_toolbar(hdc,rc,&format!("READYALERT // ENTITY   {}",row.name));let hp_pct=if row.max_hp>0{format!("{:.1}%",(row.hp.max(0)as f64*100.0/row.max_hp as f64).clamp(0.0,100.0))}else{"?%".into()};let spec=if row.subprofession_name.trim().is_empty(){profession_name(row.profession_id).to_string()}else{row.subprofession_name.clone()};
    let identity=RECT{left:8,top:TOOLBAR_H+7,right:rc.right-8,bottom:TOOLBAR_H+53};fill(hdc,&identity,rgb(24,32,38));fill(hdc,&RECT{left:identity.left,top:identity.top,right:identity.left+4,bottom:identity.bottom},rgb(66,211,190));SetTextColor(hdc,rgb(240,245,247));draw(hdc,&row.name,RECT{left:identity.left+12,top:identity.top+2,right:identity.right-250,bottom:identity.top+23},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);SetTextColor(hdc,rgb(163,180,190));draw(hdc,&format!("{}   •   UID {}   •   AS {} +{}",spec,row.uid,row.ability_score,row.illusion_break),RECT{left:identity.left+12,top:identity.top+22,right:identity.right-250,bottom:identity.bottom-2},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);SetTextColor(hdc,if row.is_dead{rgb(255,122,122)}else{rgb(199,226,220)});draw(hdc,&format!("HP {}  {}   {}",compact(row.hp as f64),hp_pct,if row.is_dead{"DEAD"}else{"ALIVE"}),RECT{left:identity.right-245,top:identity.top,right:identity.right-10,bottom:identity.bottom},DT_RIGHT|DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX);
    let cards_top=identity.bottom+7;let cards_bottom=(detail_tab_top()-74).max(cards_top+70);let gap=7;let card_w=((rc.right-16-gap*2)/3).max(160);let combat=RECT{left:8,top:cards_top,right:8+card_w,bottom:cards_bottom};let offense=RECT{left:combat.right+gap,top:cards_top,right:combat.right+gap+card_w,bottom:cards_bottom};let defense=RECT{left:offense.right+gap,top:cards_top,right:rc.right-8,bottom:cards_bottom};
    detail_card(hdc,combat,"COMBAT",&[format!("Damage   {}",compact(row.damage as f64)),format!("Healing  {}",compact(row.healing as f64)),format!("Taken    {}",compact(row.damage_taken as f64)),format!("Absorb   {}",compact(row.absorbed_damage as f64))]);
    detail_card(hdc,offense,"OFFENSE",&[format!("Magic ATK  {}",detail_attr(row,ATTR_MAGIC_ATTACK)),format!("Crit  {}   Haste  {}",format_attr_value(ATTR_CRIT,detail_attr(row,ATTR_CRIT)),format_attr_value(ATTR_HASTE,detail_attr(row,ATTR_HASTE))),format!("Luck  {}",format_attr_value(ATTR_LUCKY,detail_attr(row,ATTR_LUCKY))),format!("Mastery  {}",format_attr_value(ATTR_MASTERY,detail_attr(row,ATTR_MASTERY)))]);
    detail_card(hdc,defense,"DEFENSE",&[format!("Defense  {}",detail_attr(row,ATTR_DEFENSE_POWER)),format!("Endurance  {}",detail_attr(row,ATTR_ENDURANCE)),format!("Block  {}",block_text(row.block_pct)),format!("Deaths  {}",row.deaths)]);
    let summary_top=cards_bottom+7;let summary=RECT{left:8,top:summary_top,right:rc.right-8,bottom:detail_tab_top()-7};fill(hdc,&summary,rgb(19,25,30));SetTextColor(hdc,rgb(112,205,190));draw(hdc,&format!("{} SUMMARY",detail_mode_label(state.mode).to_ascii_uppercase()),RECT{left:summary.left+10,top:summary.top,right:summary.left+155,bottom:summary.top+21},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX);let values:Vec<String>=match state.mode{DetailMode::Damage=>{let boss_share=if row.damage>0{row.boss_damage as f64*100.0/row.damage as f64}else{0.0};vec![format!("Total {}",compact(row.damage as f64)),format!("Active/s {}",compact(row.active_dps)),format!("Encounter/s {}",compact(rate(row.damage,state.encounter_ms))),format!("Boss {:.1}%",boss_share)]},DetailMode::Heal=>{let eff=if row.healing>0{row.effective_healing as f64*100.0/row.healing as f64}else{0.0};vec![format!("Total {}",compact(row.healing as f64)),format!("Active/s {}",compact(row.active_hps)),format!("Encounter/s {}",compact(rate(row.healing,state.encounter_ms))),format!("Effective {:.1}%",eff)]},DetailMode::Tank=>vec![format!("Taken {}",compact(row.damage_taken as f64)),format!("Active/s {}",compact(row.active_dtps)),format!("Encounter/s {}",compact(rate(row.damage_taken,state.encounter_ms))),format!("Biggest {}",compact(detail_biggest_hit(row)as f64))],DetailMode::Buffs=>vec![format!("Tracked {}",row.buff_uptimes.len()),format!("Encounter {}",format_time(state.encounter_ms)),format!("Top {}",row.buff_uptimes.first().map(|b|b.name.as_str()).unwrap_or("—")),String::new()],DetailMode::Deaths=>vec![format!("Deaths {}",row.deaths),format!("Recaps {}",row.death_recaps.len()),format!("Encounter {}",format_time(state.encounter_ms)),"LAST HIT marks final recorded damage".into()]};let value_left=summary.left+160;let slot=((summary.right-value_left)/4).max(80);SetTextColor(hdc,rgb(224,232,236));for(i,text)in values.iter().enumerate(){draw(hdc,text,RECT{left:value_left+i as i32*slot,top:summary.top,right:(value_left+(i as i32+1)*slot-5).min(summary.right-5),bottom:summary.bottom},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);}
    let tab_top=detail_tab_top();paint_tab(hdc,4,tab_top,67,"Damage",state.mode==DetailMode::Damage);paint_tab(hdc,76,tab_top,70,"Healing",state.mode==DetailMode::Heal);paint_tab(hdc,151,tab_top,70,"Taken",state.mode==DetailMode::Tank);paint_tab(hdc,226,tab_top,70,"Buffs",state.mode==DetailMode::Buffs);paint_tab(hdc,301,tab_top,70,"Deaths",state.mode==DetailMode::Deaths);
    let saved=SaveDC(hdc);IntersectClipRect(hdc,0,281,rc.right,rc.bottom);SetViewportOrgEx(hdc,-state.hscroll,0,null_mut());let mut virtual_rc=rc;virtual_rc.left=state.hscroll;virtual_rc.right=state.hscroll+rc.right;match state.mode{DetailMode::Damage|DetailMode::Heal=>paint_detail_skills(hdc,virtual_rc,state),DetailMode::Tank=>paint_detail_taken(hdc,virtual_rc,state),DetailMode::Buffs=>paint_detail_buffs(hdc,virtual_rc,state),DetailMode::Deaths=>paint_detail_deaths(hdc,virtual_rc,state)}RestoreDC(hdc,saved);EndPaint(hwnd,&ps);
}


fn profession_accent_color(id:i32)->u32{match id{1=>hex_color(0xD1A8FF),2=>hex_color(0x73DAD3),3=>hex_color(0xFF9D78),4=>hex_color(0x70CCEA),5=>hex_color(0x9DDC78),8=>hex_color(0xD8C68E),9=>hex_color(0xE0C274),11=>hex_color(0xF0CD66),12=>hex_color(0x8ED2DA),13=>hex_color(0xE99AAF),14=>hex_color(0x75B6E8),15=>hex_color(0xF0A47A),_=>rgb(176,190,208)}}
fn spec_accent_color(row:&DpsRow)->u32{let spec=row.subprofession_name.trim().to_ascii_lowercase();match spec.as_str(){"iaido"|"iai"=>hex_color(0xCDA5FF),"moonstrike"=>hex_color(0x9F8CFF),"icicle"|"ice spear"=>hex_color(0x7DE3DB),"frostbeam"|"crystal"=>hex_color(0x5CC7D1),"formless"|"voidflame"=>hex_color(0xFFAD72),"crimson"|"blazecrimson"=>hex_color(0xFF7F6B),"vanguard"=>hex_color(0x69D5E2),"skyward"=>hex_color(0x6CAFE8),"smite"=>hex_color(0xA6E178),"lifebind"=>hex_color(0x72C985),"earthfort"=>hex_color(0xE1BE69),"block"=>hex_color(0xC7A96B),"wildpack"|"taming"=>hex_color(0xD9CF70),"falconry"=>hex_color(0xF0B84E),"recovery"=>hex_color(0x99D3D7),"shield"=>hex_color(0x80AFC3),"dissonance"=>hex_color(0xE893AF),"concerto"=>hex_color(0xE47D8A),_=>profession_accent_color(row.profession_id)}}
unsafe fn paint_dps_secondary_colored(hdc:HDC,row:&DpsRow,text:&str,r:RECT){let spec=dps_spec(row);if spec.is_empty()||!text.starts_with(&spec){SetTextColor(hdc,crate::ui_modern::BPSR_TEXT_SECONDARY);draw(hdc,text,r,DT_RIGHT|DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);return;}let full_w=dps_text_width(hdc,text,identity_text_px(text,false));let left=(r.right-full_w).max(r.left);let spec_w=dps_text_width(hdc,&spec,identity_text_px(&spec,false)).min((r.right-left).max(0));SetTextColor(hdc,spec_accent_color(row));draw(hdc,&spec,RECT{left,top:r.top,right:(left+spec_w).min(r.right),bottom:r.bottom},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);let suffix=&text[spec.len()..];if !suffix.is_empty()&&left+spec_w<r.right{SetTextColor(hdc,crate::ui_modern::BPSR_TEXT_SECONDARY);draw(hdc,suffix,RECT{left:left+spec_w,top:r.top,right:r.right,bottom:r.bottom},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);}}

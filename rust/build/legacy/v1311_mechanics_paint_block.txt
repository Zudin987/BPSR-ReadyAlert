unsafe fn paint_grouped_mechanic_label(hdc:HDC,state:&State,row:&GroupedMechanicRow,r:RECT){
    let right=r.right-75;let regular=if state.font.is_null(){GetStockObject(DEFAULT_GUI_FONT)}else{state.font};let bold={let font=dps_cached_font(state.scale_percent,true);if font.is_null(){regular}else{font}};let old=SelectObject(hdc,regular);let mut x=r.left+10;
    if row.targets.is_empty(){SetTextColor(hdc,crate::ui_modern::BPSR_TEXT);draw(hdc,&row.label,RECT{left:x,top:r.top,right,bottom:r.bottom},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);SelectObject(hdc,old);return;}
    let prefix=format!("{}  •  ",row.label);SetTextColor(hdc,crate::ui_modern::BPSR_TEXT);let prefix_w=dps_text_width(hdc,&prefix,identity_text_px(&prefix,false)).min((right-x).max(0));draw(hdc,&prefix,RECT{left:x,top:r.top,right:(x+prefix_w).min(right),bottom:r.bottom},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX);x+=prefix_w;
    for(index,target)in row.targets.iter().enumerate(){if x>=right{break;}let is_owner=mechanic_target_is_owner(target);SelectObject(hdc,if is_owner{bold}else{regular});SetTextColor(hdc,if is_owner{rgb(255,235,59)}else{crate::ui_modern::BPSR_TEXT});let width=dps_text_width(hdc,target,identity_text_px(target,is_owner)).min((right-x).max(0));draw(hdc,target,RECT{left:x,top:r.top,right:(x+width).min(right),bottom:r.bottom},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);x+=width;if index+1<row.targets.len()&&x<right{SelectObject(hdc,regular);SetTextColor(hdc,crate::ui_modern::BPSR_TEXT_SECONDARY);let sep=", ";let sep_w=dps_text_width(hdc,sep,10).min((right-x).max(0));draw(hdc,sep,RECT{left:x,top:r.top,right:(x+sep_w).min(right),bottom:r.bottom},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX);x+=sep_w;}}
    SelectObject(hdc,old);
}
unsafe fn paint_special_tracker_group(hdc:HDC,state:&State,items:&[(String,crate::event_tracker::TrackerDisplayRow)],r:RECT){
    if items.is_empty(){return;}fill(hdc,&RECT{left:r.left,top:r.top,right:r.left+3,bottom:r.bottom},crate::ui_modern::BPSR_WARNING);let usable_left=r.left+5;let usable_right=r.right-4;let count=items.len()as i32;let width=((usable_right-usable_left)/count.max(1)).max(1);let regular=if state.font.is_null(){GetStockObject(DEFAULT_GUI_FONT)}else{state.font};let bold={let font=dps_cached_font(state.scale_percent,true);if font.is_null(){regular}else{font}};let old=SelectObject(hdc,bold);
    for(index,(name,row))in items.iter().enumerate(){let left=usable_left+index as i32*width;let right=if index+1==items.len(){usable_right}else{left+width};if index>0{fill(hdc,&RECT{left,top:r.top+4,right:left+1,bottom:r.bottom-4},crate::ui_modern::DARK_BORDER);}let mid=r.top+(r.bottom-r.top)/2;SetTextColor(hdc,crate::ui_modern::BPSR_TEXT);draw(hdc,name,RECT{left:left+2,top:r.top,right:right-2,bottom:mid+2},DT_CENTER|DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);SelectObject(hdc,regular);let(timer,color)=match row.remaining_ms{Some(ms)=>(format!("{:.1}s",ms.max(0)as f64/1000.0),timer_urgency_color(ms)),None=>("ACTIVE".into(),crate::ui_modern::BPSR_ACCENT)};SetTextColor(hdc,color);draw(hdc,&timer,RECT{left:left+2,top:mid-1,right:right-2,bottom:r.bottom},DT_CENTER|DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);SelectObject(hdc,bold);}
    SelectObject(hdc,old);
}
unsafe fn paint_mechanics(hdc:HDC,rc:RECT,state:&State){
    let features=state.features.read().map(|x|x.clone()).unwrap_or_default();let tracked:Vec<_>=features.mechanic_attributes.tracked.iter().filter_map(|id|state.mechanics.tracked_attributes.iter().find(|attr|attr.attr_id==*id)).collect();let mut top=TOOLBAR_H+5;
    if !tracked.is_empty(){let rows=mechanic_attr_rows(tracked.len());for row_i in 0..rows{let start=row_i*4;let end=(start+4).min(tracked.len());let count=end.saturating_sub(start);if count==0{continue;}let strip=RECT{left:6,top,right:rc.right-6,bottom:top+MECH_ATTR_H-3};fill(hdc,&strip,crate::ui_modern::DARK_SURFACE);let width=((strip.right-strip.left)/count as i32).max(60);for(i,attr)in tracked[start..end].iter().enumerate(){let left=strip.left+i as i32*width;let zone=RECT{left:left+2,top:strip.top,right:(left+width-2).min(strip.right),bottom:strip.bottom};let mid=zone.top+(zone.bottom-zone.top)/2;SetTextColor(hdc,crate::ui_modern::BPSR_MUTED);draw(hdc,mechanic_attr_overlay_label(attr.attr_id,attr.label.trim_end_matches(" %")),RECT{left:zone.left,top:zone.top,right:zone.right,bottom:mid+1},DT_CENTER|DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);SetTextColor(hdc,attr_color(attr.attr_id));draw(hdc,&format_attr_value(attr.attr_id,attr.value),RECT{left:zone.left,top:mid-1,right:zone.right,bottom:zone.bottom},DT_CENTER|DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);}top+=MECH_ATTR_H;}}
    let consumables=RECT{left:6,top,right:rc.right-6,bottom:top+MECH_CONSUMABLE_H-4};fill(hdc,&consumables,crate::ui_modern::DARK_SURFACE);paint_consumable_status(hdc,"Food",state.mechanics.food.as_ref(),RECT{left:consumables.left+8,top:consumables.top,right:consumables.right-8,bottom:consumables.top+20});paint_consumable_status(hdc,"Serum",state.mechanics.serum.as_ref(),RECT{left:consumables.left+8,top:consumables.top+20,right:consumables.right-8,bottom:consumables.bottom});top+=MECH_CONSUMABLE_H;
    let now=now_ms();let rows=ordered_mechanic_rows(state,now);let total=rows.len();let visible=(((rc.bottom-top)/MECH_ROW_H).max(0)as usize).min(MAX_MECHANIC_ROWS_VISIBLE);
    if total==0{crate::ui_modern::qa_draw_empty_state(hdc,RECT{left:6,top,right:rc.right-6,bottom:rc.bottom},"No active mechanics","Mechanics and tracker events appear here when detected.");return;}
    for screen_i in 0..visible{let index=state.scroll+screen_i;if index>=total{break;}let y=top+screen_i as i32*MECH_ROW_H;let r=RECT{left:6,top:y,right:rc.right-8,bottom:y+MECH_ROW_H-3};fill(hdc,&r,if screen_i%2==0{crate::ui_modern::DARK_RAISED}else{crate::ui_modern::DARK_SURFACE});match &rows[index]{MechanicDisplayRow::Tracker(row)=>{fill(hdc,&RECT{left:r.left,top:r.top,right:r.left+3,bottom:r.bottom},crate::ui_modern::BPSR_WARNING);let detail=if row.detail.trim().is_empty(){String::new()}else{format!("  •  {}",row.detail)};SetTextColor(hdc,crate::ui_modern::BPSR_TEXT);draw(hdc,&format!("Tracker  •  {}{}",row.label,detail),RECT{left:r.left+10,top:r.top,right:r.right-92,bottom:r.bottom},DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX|DT_END_ELLIPSIS);let(timer,color)=match row.remaining_ms{Some(ms)=>(format!("{:.1}s",ms.max(0)as f64/1000.0),timer_urgency_color(ms)),None=>("ACTIVE".into(),crate::ui_modern::BPSR_ACCENT)};SetTextColor(hdc,color);draw(hdc,&timer,RECT{left:r.right-88,top:r.top,right:r.right-8,bottom:r.bottom},DT_RIGHT|DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX);},MechanicDisplayRow::TrackerGroup(items)=>paint_special_tracker_group(hdc,state,items,r),MechanicDisplayRow::Mechanic(row)=>{let accent=if row.priority>=3{crate::ui_modern::BPSR_DANGER}else{crate::ui_modern::BPSR_ACCENT};fill(hdc,&RECT{left:r.left,top:r.top,right:r.left+3,bottom:r.bottom},accent);paint_grouped_mechanic_label(hdc,state,row,r);let remaining=if row.persistent||row.expires_unix_ms<=0{None}else{Some(row.expires_unix_ms.saturating_sub(now))};SetTextColor(hdc,remaining.map(timer_urgency_color).unwrap_or(crate::ui_modern::BPSR_ACCENT));draw(hdc,&mechanic_timer(row.expires_unix_ms,row.persistent,now),RECT{left:r.right-70,top:r.top,right:r.right-8,bottom:r.bottom},DT_RIGHT|DT_VCENTER|DT_SINGLELINE|DT_NOPREFIX);}}}
    paint_scrollbar(hdc,rc,total,visible,state.scroll,top);
}
